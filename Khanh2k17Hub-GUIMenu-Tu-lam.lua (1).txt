local function createCorner(parent, radius)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, radius or 8)
    corner.Parent = parent
    return corner

local function createLayout(parent, padding)
    local layout = Instance.new("UIListLayout")
    layout.Padding = UDim.new(0, padding or 6)
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    layout.Parent = parent
    return layout
end

    stroke.Thickness = thickness or 1

    return stroke

--==================================================
-- DRAG SYSTEM
--==================================================

local function beginDrag(input, mode)
    if dragging then
        return
    end

    if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    dragging = true
    dragMoved = false
    dragMode = mode
    dragInput = input
    dragStart = input.Position
    startPosition = Main.Position
end

local function updateDrag(input)
    if not dragging or not dragStart then
        return
    end

    if input.UserInputType == Enum.UserInputType.Touch
        and dragInput ~= input then
        return
    end

    if input.UserInputType ~= Enum.UserInputType.MouseMovement
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    local delta = input.Position - dragStart

    if delta.Magnitude >= DRAG_THRESHOLD then
        dragMoved = true
    end

    if dragMoved then
        Main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end

local function expandMenu()
    if not minimized then
        return
    end

    minimized = false

    MiniStroke.Enabled = false
    MainStroke.Enabled = true

    TweenService:Create(
        Main,
        TweenInfo.new(
            0.2,
            Enum.EasingStyle.Quad,
            Enum.EasingDirection.Out
        ),
        {
            Size = UDim2.fromOffset(GUI_WIDTH, GUI_HEIGHT)
        }
    ):Play()

    Header.Size = UDim2.new(1, 0, 0, 44)

    Toggle.AnchorPoint = Vector2.new(0, 0)
    Toggle.Position = UDim2.new(1, -72, 0, 7)
    Toggle.Size = UDim2.fromOffset(32, 30)
    Toggle.Text = "-"
    Toggle.TextSize = 24
    Toggle.BackgroundTransparency = 0
    Toggle.ZIndex = 5

    Title.Visible = true
    Close.Visible = true
    Sidebar.Visible = true
    Content.Visible = true
end

            0.2,

        }
    ):Play()
end

--==================================================
-- HEADER DRAG
--==================================================

Header.InputBegan:Connect(function(input)
    if minimized then
        return
    end

    local relativeX =
        input.Position.X - Header.AbsolutePosition.X

    if relativeX > Header.AbsoluteSize.X - 110 then
        return
    end

    beginDrag(input, "header")
end)

Toggle.InputBegan:Connect(function(input)
    if minimized then
        beginDrag(input, "mini")
    else
        beginDrag(input, "toggle")
    end
end)

UIS.InputChanged:Connect(function(input)
    updateDrag(input)
end)

UIS.InputEnded:Connect(function(input)
    if not dragging then
        return
    end

    if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    local oldMode = dragMode
    local wasMoved = dragMoved

    dragging = false
    dragMoved = false
    dragMode = nil
    dragInput = nil
    dragStart = nil
    startPosition = nil

    if oldMode == "mini" and not wasMoved and minimized then
        expandMenu()
    end
end)

function Window:AddTab(config)
    config = config or {}

    local name = config.Title or "Tab"

    local Tab = {
        Title = name,
        Toggles = {},
        Buttons = {}
    }

    local TabButton = Instance.new("TextButton")
    TabButton.Name = name .. "Tab"
    TabButton.Size = UDim2.new(1, 0, 0, 35)
    TabButton.BackgroundColor3 = Color3.fromRGB(35, 35, 48)
    TabButton.Text = name
    TabButton.TextColor3 = Color3.fromRGB(200, 200, 215)
    TabButton.TextSize = 13
    TabButton.Font = Enum.Font.GothamSemibold
    TabButton.BorderSizePixel = 0
    TabButton.AutoButtonColor = false
    TabButton.Parent = SidebarList

    createCorner(TabButton, 7)

    local Container = Instance.new("Frame")
    Container.Name = name .. "Container"
    Container.Size = UDim2.new(1, -5, 0, 0)
    Container.AutomaticSize = Enum.AutomaticSize.Y
    Container.BackgroundTransparency = 1
    Container.Visible = false
    Container.Parent = PageScroll

    local ContainerLayout = createLayout(Container, 7)

    ContainerLayout:GetPropertyChangedSignal(
        "AbsoluteContentSize"
    ):Connect(function()
        if CurrentTab == Tab then
            PageScroll.CanvasSize = UDim2.new(
                0, 0,
                0, ContainerLayout.AbsoluteContentSize.Y + 10
            )
        end
    end)

    Tab.Button = TabButton
    Tab.Container = Container

    local function selectTab()
        CurrentTab = Tab
        PageTitle.Text = name

        for _, item in ipairs(Tabs) do
            item.Button.BackgroundColor3 =
                Color3.fromRGB(35, 35, 48)

            item.Button.TextColor3 =
                Color3.fromRGB(200, 200, 215)

            item.Container.Visible = false
        end

        TabButton.BackgroundColor3 =
            Color3.fromRGB(50, 80, 60)

        TabButton.TextColor3 = Color3.new(1, 1, 1)

        Container.Visible = true

        PageScroll.CanvasPosition = Vector2.new(0, 0)

        PageScroll.CanvasSize = UDim2.new(
            0, 0,
            0, ContainerLayout.AbsoluteContentSize.Y + 10
        )
    end

    Tab.Select = selectTab

    TabButton.Activated:Connect(selectTab)

    --==================================================
    -- API: Tab:AddToggle
    --==================================================

    function Tab:AddToggle(Flag, options)
        options = options or {}

        local title = options.Title or Flag
        local callback = options.Callback or function() end
        local enabled = options.Default == true

        local Row = Instance.new("Frame")
        Row.Name = tostring(Flag) .. "Toggle"
        Row.Size = UDim2.new(1, -5, 0, 52)
        Row.BackgroundColor3 = Color3.fromRGB(32, 32, 44)
        Row.BorderSizePixel = 0
        Row.Parent = Container

        createCorner(Row, 8)

        local Label = Instance.new("TextLabel")
        Label.Name = "Title"
        Label.Size = UDim2.new(1, -75, 1, 0)
        Label.Position = UDim2.fromOffset(12, 0)
        Label.BackgroundTransparency = 1
        Label.Text = title
        Label.TextColor3 = Color3.new(1, 1, 1)
        Label.TextSize = 14
        Label.Font = Enum.Font.GothamSemibold
        Label.TextXAlignment = Enum.TextXAlignment.Left
        Label.Parent = Row

        local Switch = Instance.new("TextButton")
        Switch.Name = "Switch"
        Switch.Size = UDim2.fromOffset(46, 25)
        Switch.Position = UDim2.new(1, -58, 0.5, -12)
        Switch.BackgroundColor3 = Color3.fromRGB(65, 65, 75)
        Switch.Text = ""
        Switch.AutoButtonColor = false
        Switch.BorderSizePixel = 0
        Switch.Parent = Row

        createCorner(Switch, 20)

        local Knob = Instance.new("Frame")
        Knob.Name = "Knob"
        Knob.Size = UDim2.fromOffset(19, 19)
        Knob.Position = UDim2.new(0, 3, 0.5, -9)
        Knob.BackgroundColor3 = Color3.new(1, 1, 1)
        Knob.BorderSizePixel = 0
        Knob.Parent = Switch

        createCorner(Knob, 20)

        local function SetValue(value, fireCallback)
            enabled = value == true

            local color = enabled
                and Color3.fromRGB(34, 197, 94)
                or Color3.fromRGB(65, 65, 75)

            local position = enabled
                and UDim2.new(1, -22, 0.5, -9)
                or UDim2.new(0, 3, 0.5, -9)

            TweenService:Create(
                Switch,
                TweenInfo.new(0.18),
                {
                    BackgroundColor3 = color
                }
            ):Play()

            TweenService:Create(
                Knob,
                TweenInfo.new(0.18),
                {
                    Position = position
                }
            ):Play()

            if fireCallback ~= false then
                local ok, err = pcall(callback, enabled)

                if not ok then
                    warn("KhanhHub Toggle Error:", err)
                end
            end
        end

        Switch.Activated:Connect(function()
            SetValue(not enabled, true)
        end)

        local ToggleAPI = {}

        function ToggleAPI:SetValue(value)
            SetValue(value, true)
        end

        function ToggleAPI:GetValue()
            return enabled
        end

        function ToggleAPI:Destroy()
            Row:Destroy()
        end

        table.insert(Tab.Toggles, ToggleAPI)

        SetValue(enabled, false)

        return ToggleAPI
    end

    --==================================================
    -- API: Tab:AddButton
    --==================================================

    function Tab:AddButton(options)
        options = options or {}

        local Button = Instance.new("TextButton")
        Button.Name = options.Title or "Button"
        Button.Size = UDim2.new(1, -5, 0, 42)
        Button.BackgroundColor3 = Color3.fromRGB(45, 45, 62)
        Button.Text = options.Title or "Button"
        Button.TextColor3 = Color3.new(1, 1, 1)
        Button.TextSize = 14
        Button.Font = Enum.Font.GothamSemibold
        Button.BorderSizePixel = 0
        Button.Parent = Container

        createCorner(Button, 8)

        Button.Activated:Connect(function()
            if options.Callback then
                local ok, err = pcall(options.Callback)

                if not ok then
                    warn("KhanhHub Button Error:", err)
                end
            end
        end)

        table.insert(Tab.Buttons, Button)

        return Button
    end

    table.insert(Tabs, Tab)

    if #Tabs == 1 then
        selectTab()
    end

    return Tab
end
